Please wait."; resultImg.style.display = "none"; // ВНИМАНИЕ: Вставь свой НОВЫЙ токен строго между кавычками. // Убедись, что нет лишних пробелов. const apiToken = "hf_ВСТАВЬ_СВОЙ_НОВЫЙ_ТОКЕН_ТУТ"; try { const response = await fetch( "https://api-inference.huggingface.co/models/shallowdream204/bitdance", { method: "POST", headers: { "Authorization": "Bearer " + apiToken.trim(), "Content-Type": "application/json" }, body: JSON.stringify({ inputs: promptInput }), } ); // Если модель загружается (503), нужно подождать if (response.status === 503) { const data = await response.json(); statusText.innerText = "🛰️ Model is waking up...
